```typescript
export const WINDOW_7D_WIDTH_DAYS = 7;   // inclusive width, NOT an offset — mutation test flips this to 6
export const WINDOW_30D_WIDTH_DAYS = 30;

export type WindowValue = 'today' | '7d' | '30d' | '';

export interface FilterParams {
  family: string[];   // parsed from comma-separated ?family=
  region: string[];   // parsed from comma-separated ?region= — strings, never Number()
  window: WindowValue; // '' means "all dates" (absence of the param), never the literal 'all'
  q: string;           // comuna search text, raw (not normalized) — mirrors ?q= on communes/index.astro
}

export function norm(s: string): string;
// s.normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase() — algorithm copied
// verbatim from communes/index.astro:175-176, defined ONCE here (F-27), never re-declared
// inline in a page <script>.

export function parseFilterParams(search: string): FilterParams;
// search = location.search (or any "?a=b&c=d" string, leading "?" optional).
// Unknown/invalid `window` values (anything other than 'today'|'7d'|'30d') MUST
// degrade to '' (all dates) — never throw (RESEARCH.md Security Domain: hostile
// query-param values must degrade to "no match"/safe default, never throw).

export function serializeFilterParams(params: FilterParams): string;
// Returns a query string WITHOUT the leading '?'. Omits a dimension entirely when
// it is empty (empty family[]/region[] array, window==='', q==='') so the URL
// stays minimal — round-trips through parseFilterParams to an equal FilterParams.

export function computeNewestDate(dates: string[]): string | null;
// Port of newsFacets.ts:180-183's loop — max of a plain YYYY-MM-DD string array
// (lexicographic order === chronological order for this format). Returns null
// for an empty array.

export function matchesWindow(dateStr: string, newestDate: string | null, window: WindowValue): boolean;
// window==='' always matches (true), regardless of newestDate. If newestDate is
// null and window!=='' , returns false (never throws). Boundaries are INCLUSIVE
// both ends, ported verbatim from newsFacets.ts:179-216 using WINDOW_7D_WIDTH_DAYS-1
// and WINDOW_30D_WIDTH_DAYS-1 as the day offsets (width 7 = anchor + 6 prior days).

export interface CardFilterState {
  family: string;       // '' when the card has no family
  regionId: string;     // '' when unresolved
  date: string;          // ISO YYYY-MM-DD
  communeNorm: string;   // '' when no commune resolved; already norm()-ed at build time
}

export function cardMatchesFilters(
  card: CardFilterState,
  active: FilterParams,
  newestDate: string | null,
  excludeDimension?: 'family' | 'region' | 'window'
): boolean;
// Applies every active filter EXCEPT excludeDimension (F-28 self-exclusion). The
// comuna query (q) is cross-cutting and is ALWAYS applied regardless of
// excludeDimension (per F-28: "the comuna query is a cross-cutting filter applied
// in all three passes"). family/region use `.includes()` against active arrays
// (empty active array = dimension not filtering = always matches).

export function computeFacetCounts(
  cards: CardFilterState[],
  active: FilterParams,
  newestDate: string | null,
  dimension: 'family' | 'region' | 'window',
  keys: string[]
): Record<string, number>;
// For each key in `keys`, counts cards where card[dimension-equivalent field] === key
// AND cardMatchesFilters(card, active, newestDate, dimension) is true (self-exclusion).
// For dimension 'window', keys are 'today'|'7d'|'30d' and the per-key match uses
// matchesWindow(card.date, newestDate, key) instead of equality.
```
