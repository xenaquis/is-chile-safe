## .github/workflows/news-pipeline.yml (replace entire file with exactly this)

```yaml
name: News Pipeline

on:
  schedule:
    - cron: '0 */6 * * *'
  workflow_dispatch:

concurrency:
  group: news-pipeline
  cancel-in-progress: false

jobs:
  scrape:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    permissions:
      contents: write
      issues: write
    # CR-01: CF_HOOK MUST be job-level — a step-level env var is NOT in scope for
    # that same step's `if:` condition, so the deploy gate would never fire.
    env:
      CF_HOOK: ${{ secrets.CF_DEPLOY_HOOK_URL }}

    steps:
      # fetch-depth: 0 (CRON-06) — the retry loop's `git rebase origin/master` needs
      # real history to compute a merge-base; a shallow (depth-1) checkout can leave
      # the rebase without a common ancestor if origin has moved forward.
      - uses: actions/checkout@v7
        with:
          fetch-depth: 0

      # CRON-01 (F-76): CF_DEPLOY_HOOK_URL is REQUIRED here, not optional — Cloudflare
      # Pages automatic production builds are disabled (see deploy-on-code.yml header),
      # so an empty hook means every commit from this point silently never reaches
      # production. Hard-fail before any real work runs.
      - name: Guard required secrets
        env:
          OPENROUTER_API_KEY: ${{ secrets.OPENROUTER_API_KEY }}
          DEEPSEEK_API_KEY: ${{ secrets.DEEPSEEK_API_KEY }}
          CF_DEPLOY_HOOK_URL: ${{ secrets.CF_DEPLOY_HOOK_URL }}
        run: bash .github/scripts/require-env.sh OPENROUTER_API_KEY DEEPSEEK_API_KEY CF_DEPLOY_HOOK_URL

      - uses: actions/setup-python@v7
        with:
          python-version: '3.12'

      - name: Install pipeline deps
        run: pip install -r pipeline/requirements.txt

      - name: Run news scraper
        env:
          OPENROUTER_API_KEY: ${{ secrets.OPENROUTER_API_KEY }}
          DEEPSEEK_API_KEY: ${{ secrets.DEEPSEEK_API_KEY }}
        run: python pipeline/scrape_news.py

      - name: Commit data if changed
        id: commit
        run: |
          git config user.name  "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add data/
          if git diff --staged --quiet; then
            echo "changed=false" >> "$GITHUB_OUTPUT"
          else
            git commit -m "data: auto-update news [skip ci]"
            pushed=0
            for i in 1 2 3 4 5; do
              if git push; then
                pushed=1
                break
              fi
              echo "push rejected, attempt $i of 5 — fetching and rebasing (CRON-06)"
              git fetch origin master
              if ! git rebase origin/master; then
                echo "::error::rebase conflict during data commit — aborting, a human must resolve"
                git rebase --abort
                exit 1
              fi
              sleep $((i * 3))
            done
            if [ "$pushed" -ne 1 ]; then
              echo "::error::push failed after 5 attempts"
              exit 1
            fi
            echo "changed=true" >> "$GITHUB_OUTPUT"
          fi

      # CRON-03 (F-78): canonical deploy-hook invocation, byte-identical across
      # news-pipeline.yml, cead-scraper.yml, deploy-on-code.yml (grep-provable, see
      # verify block). CRON-01 (F-76): the previous CF_HOOK-blank skip condition is
      # removed here — the guard
      # step above already made an empty hook fatal, so reaching this step guarantees
      # CF_HOOK is set.
      - name: Trigger Cloudflare Pages deploy
        if: steps.commit.outputs.changed == 'true'
        run: curl -X POST --fail --silent --show-error --retry 3 --retry-all-errors --max-time 60 "$CF_HOOK"

      - name: Alert via GitHub Issue on failure
        if: failure()
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh label create pipeline-failure-news --color e11d48 --description "News Pipeline failure alert" --force
          OPEN_ISSUE=$(gh issue list --label pipeline-failure-news --state open --json number --jq '.[0].number // empty')
          RUN_URL="${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}"
          if [ -n "$OPEN_ISSUE" ]; then
            gh issue comment "$OPEN_ISSUE" --body "News Pipeline failed again: $RUN_URL"
          else
            gh issue create \
              --label pipeline-failure-news \
              --label pipeline-failure \
              --title "[pipeline] News Pipeline failed on $(date -u +%F)" \
              --body "Pipeline run failed: $RUN_URL"
          fi
```

# cead-scraper.yml
#
# CRON-05: THIS CRON IS EXPECTED TO FAIL ON GITHUB ACTIONS.
# ------------------------------------------------------------
# CEAD (Centro de Estudios y Analisis del Delito) blocks GitHub Actions runner IP
# ranges with HTTP 403. The real quarterly scrape runs LOCALLY on a maintainer's
# machine (see the `cead-scraper-quarterly-local` project memory) and is committed
# and pushed by hand. This scheduled workflow exists ONLY as a calendar reminder:
# when `pipeline/scrape_cead.py` fails with a 403 here, that is the expected,
# documented outcome — NOT a regression, and NOT something to "fix" by retrying
# from Actions. The failure-alert issue body below says this explicitly so a
# maintainer unfamiliar with the history does not mistake it for a real bug.

name: CEAD Scraper (Quarterly)

on:
  schedule:
    - cron: '0 3 1 1,4,7,10 *'
  workflow_dispatch:

concurrency:
  group: cead-scraper
  cancel-in-progress: false

jobs:
  scrape:
    runs-on: ubuntu-latest
    timeout-minutes: 60
    permissions:
      contents: write
      issues: write
    # CR-01: CF_HOOK MUST be job-level — a step-level env var is NOT in scope for
    # that same step's `if:` condition, so the deploy gate would never fire.
    env:
      CF_HOOK: ${{ secrets.CF_DEPLOY_HOOK_URL }}
      PYTHONPATH: .

    steps:
      # fetch-depth: 0 (CRON-06) — see news-pipeline.yml for the same rationale.
      - uses: actions/checkout@v7
        with:
          fetch-depth: 0

      # CRON-01 (F-76): CF_DEPLOY_HOOK_URL is REQUIRED, not optional.
      - name: Guard required secrets
        env:
          CF_DEPLOY_HOOK_URL: ${{ secrets.CF_DEPLOY_HOOK_URL }}
        run: bash .github/scripts/require-env.sh CF_DEPLOY_HOOK_URL

      - uses: actions/setup-python@v7
        with:
          python-version: '3.12'

      - name: Install pipeline deps
        run: pip install -r pipeline/requirements.txt

      # CRON-05: expected to fail with HTTP 403 from GitHub Actions runner IPs — see
      # the file header comment. Not guarded with continue-on-error: a real failure
      # here (network, schema change) must still alert; it is indistinguishable in
      # the Actions UI from the expected 403, which is exactly why the failure-alert
      # body below (not the exit code) is what disambiguates it for a reader.
      - name: Run CEAD scraper
        run: python pipeline/scrape_cead.py

      - name: Build ENUSC victimization snapshot
        continue-on-error: true
        env:
          CACHE_ONLY: '1'
        run: python pipeline/snapshots/fetch_enusc_vhdv.py

      - name: Enrich commune JSONs with ENUSC VHDV
        continue-on-error: true
        run: python pipeline/build_enusc_enrichment.py

      - name: Build composite index
        continue-on-error: true
        run: python pipeline/build_composite_index.py

      - name: Build map payload
        continue-on-error: true
        run: python pipeline/build_map_payload.py

      - name: Commit data if changed
        id: commit
        run: |
          git config user.name  "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add data/
          if git diff --staged --quiet; then
            echo "changed=false" >> "$GITHUB_OUTPUT"
          else
            git commit -m "data: auto-update CEAD [skip ci]"
            pushed=0
            for i in 1 2 3 4 5; do
              if git push; then
                pushed=1
                break
              fi
              echo "push rejected, attempt $i of 5 — fetching and rebasing (CRON-06)"
              git fetch origin master
              if ! git rebase origin/master; then
                echo "::error::rebase conflict during data commit — aborting, a human must resolve"
                git rebase --abort
                exit 1
              fi
              sleep $((i * 3))
            done
            if [ "$pushed" -ne 1 ]; then
              echo "::error::push failed after 5 attempts"
              exit 1
            fi
            echo "changed=true" >> "$GITHUB_OUTPUT"
          fi

      - name: Trigger Cloudflare Pages deploy
        if: steps.commit.outputs.changed == 'true'
        run: curl -X POST --fail --silent --show-error --retry 3 --retry-all-errors --max-time 60 "$CF_HOOK"

      - name: Alert via GitHub Issue on failure
        if: failure()
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh label create pipeline-failure-cead --color e11d48 --description "CEAD Scraper failure alert" --force
          OPEN_ISSUE=$(gh issue list --label pipeline-failure-cead --state open --json number --jq '.[0].number // empty')
          RUN_URL="${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}"
          if [ -n "$OPEN_ISSUE" ]; then
            gh issue comment "$OPEN_ISSUE" --body "CEAD Scraper failed again (expected quarterly reminder if this is an HTTP 403 — see workflow header — otherwise a real regression): $RUN_URL"
          else
            gh issue create \
              --label pipeline-failure-cead \
              --label pipeline-failure \
              --title "[pipeline] CEAD Scraper failed on $(date -u +%F)" \
              --body "This is the expected quarterly reminder if the failure is an HTTP 403 — CEAD blocks GitHub Actions runner IPs. Run pipeline/scrape_cead.py locally and push. If the failure is NOT a 403, this is a real regression: $RUN_URL"
          fi
```

# deploy-on-code.yml
#
# WHY THIS EXISTS
# ---------------
# Cloudflare Pages automatic production builds are DISABLED (DEPLOYMENT.md §4) to prevent a
# rebuild loop caused by the every-6h data commits. Those data commits carry [skip ci] and
# stage ONLY data/ files, so they will NOT match `paths: site/**` — no loop is reintroduced.
# Side effect of disabling CF auto-build: pushes that contain ONLY site code never redeploy
# production; the Deploy Hook is the sole production build trigger. This workflow closes that
# gap for code pushes.
#
# CRON-01 (F-76): CF_DEPLOY_HOOK_URL is REQUIRED, not optional. An empty hook here used to
# "dry-run" (exit 0, log a message) — that is now a hard failure. A missing hook does not
# mean "nothing to do"; it means every code push from this point silently never reaches
# production, indefinitely, while this job reports green. That is the exact silent-no-op
# CRON-01 exists to abolish, and the highest-severity instance of it in the repo (this
# workflow is the SOLE production build trigger for code pushes — see above).
#
# Note: DEPLOYMENT.md documentation update (§1 and §10 workflow-cadence table) for this new
# trigger is a deliberate follow-up, out of scope here.

name: Deploy on Code Push

on:
  push:
    branches: [master]
    paths:
      - 'site/**'
      - '.github/workflows/deploy-on-code.yml'

permissions:
  contents: read

jobs:
  deploy:
    runs-on: ubuntu-latest
    # CR-01: CF_HOOK MUST be job-level — a step-level env var is NOT in scope for
    # that same step's `if:` condition, so the deploy gate would never fire.
    env:
      CF_HOOK: ${{ secrets.CF_DEPLOY_HOOK_URL }}

    steps:
      # A checkout is now needed (it was not before) so this job can reach the
      # shared guard script under .github/scripts/. This is a deliberate addition,
      # not an oversight — noted here so a future reader does not "clean it up".
      - uses: actions/checkout@v7

      - name: Guard required secrets
        env:
          CF_DEPLOY_HOOK_URL: ${{ secrets.CF_DEPLOY_HOOK_URL }}
        run: bash .github/scripts/require-env.sh CF_DEPLOY_HOOK_URL

      # CRON-03 (F-78): canonical deploy-hook invocation, byte-identical to
      # news-pipeline.yml and cead-scraper.yml.
      - name: Trigger Cloudflare Pages deploy
        run: curl -X POST --fail --silent --show-error --retry 3 --retry-all-errors --max-time 60 "$CF_HOOK"
```
