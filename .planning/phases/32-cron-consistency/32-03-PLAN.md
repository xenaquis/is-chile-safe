## .github/workflows/heartbeat.yml (create exactly this content)

```yaml
name: Pipeline Heartbeat

# CRON-02 / F-79: this workflow's ENTIRE purpose is detecting that another scheduled
# workflow has gone silent. It MUST NOT live inside the workflow it watches (a
# self-referential heartbeat cannot detect its own host's total absence). Runs on its
# own daily schedule, independent of news-pipeline.yml / r2-archive.yml /
# cead-scraper.yml.
#
# Accepted residual gap (F-79): this workflow's OWN silence is not monitored by
# anything in this repo. freshness.mjs (validator #16, run on every local
# `npm run validate` and in ci.yml) independently re-derives the news signal from
# data/, so news staleness survives even if this cron also dies. R2 archive
# staleness is the ONE signal that depends on this workflow being alive -- closing
# that fully would require an external monitor, which the ~$0 infrastructure
# constraint forbids.

on:
  schedule:
    - cron: '0 12 * * *'
  workflow_dispatch:

permissions:
  contents: read
  issues: write

jobs:
  heartbeat:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      # fetch-depth: 0 -- the CEAD check below needs full path history
      # ('git log -1 -- data/cead/'); a shallow clone would see only the tip commit
      # and could falsely report "no evidence" even when data/cead/ was updated
      # further back in history.
      - uses: actions/checkout@v7
        with:
          fetch-depth: 0

      - name: Check R2 archive heartbeat (gh run list evidence, 4-day threshold per F-79)
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          LAST_RUN=$(gh run list --workflow=r2-archive.yml --json createdAt --jq '.[0].createdAt // empty' --limit 1)
          bash .github/scripts/check-heartbeat.sh r2 4 "$LAST_RUN"

      - name: Check CEAD reminder heartbeat (git log evidence, 100-day threshold per F-79)
        run: |
          LAST_COMMIT=$(git log -1 --format=%cI -- data/cead/)
          bash .github/scripts/check-heartbeat.sh cead 100 "$LAST_COMMIT"

      - name: Alert via GitHub Issue on failure
        if: failure()
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh label create pipeline-failure-heartbeat --color e11d48 --description "Pipeline Heartbeat failure alert" --force
          OPEN_ISSUE=$(gh issue list --label pipeline-failure-heartbeat --state open --json number --jq '.[0].number // empty')
          RUN_URL="${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}"
          if [ -n "$OPEN_ISSUE" ]; then
            gh issue comment "$OPEN_ISSUE" --body "Pipeline Heartbeat failed again: $RUN_URL"
          else
            gh issue create \
              --label pipeline-failure-heartbeat \
              --label pipeline-failure \
              --title "[pipeline] Heartbeat detected a stalled cron on $(date -u +%F)" \
              --body "Heartbeat run failed: $RUN_URL -- check which step (R2 archive or CEAD reminder) failed in the run log."
          fi
```
